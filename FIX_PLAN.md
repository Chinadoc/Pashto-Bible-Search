## Part 5: Automation & Safeguards
**Goal:** Add tests to prevent regressions and ensure filtering works correctly.

### Steps:

1. **API Integration Test**
   ```typescript
   // tests/api/search.test.ts
   describe('Search API - Verb Recognition', () => {
     it('should recognize وهل as verb', async () => {
       const response = await POST('/api/search', {
         query: 'وهل',
         includeRelated: true,
       });
       expect(response.relatedForms.posGuess).toBe('verb');
       expect(response.relatedForms.forms.verbs.length).toBeGreaterThan(0);
     });

     it('should include 1st person forms for وهل', async () => {
       const response = await POST('/api/search', {
         query: 'وهل',
         includeRelated: true,
       });
       const verbForms = response.relatedForms.forms.verbs;
       const firstPersonForms = verbForms.filter(v => 
         v.label.includes('1sg') || v.label.includes('1pl')
       );
       expect(firstPersonForms.length).toBeGreaterThan(0);
     });

     it('should filter variants correctly', async () => {
       const response = await POST('/api/search', {
         query: 'وهل',
         variants: ['وهم', 'وهو'], // Only 1st person forms
         includeRelated: true,
       });
       // Verify results contain only these forms
       expect(response.results.length).toBeLessThan(280);
     });
   });
   ```

2. **Client Unit Test**
   ```typescript
   // tests/client/filtering.test.tsx
   describe('Verb Filtering', () => {
     it('should filter verb variants by person', () => {
       const verbs = [
         { form: 'وهم', label: '1sg Present' },
         { form: 'وهو', label: '1pl Present' },
         { form: 'وهې', label: '2sg Present' },
       ];
       const filters = {
         person: ['1st'],
         tense: ['all'],
         aspect: ['all'],
         mood: ['all'],

       const filtered = filterVerbVariantsMulti(verbs, filters);
       expect(filtered.length).toBe(2);
       expect(filtered.map(v => v.form)).toEqual(['وهم', 'وهو']);
     });
   });
   ```
